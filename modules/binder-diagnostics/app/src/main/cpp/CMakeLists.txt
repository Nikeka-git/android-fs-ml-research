cmake_minimum_required(VERSION 3.22.1)
project(production_core LANGUAGES CXX)

if(ANDROID)
    # The historical library name is retained; this is local simulation plus
    # official BINDER_VERSION diagnostics, with no exploitation functionality.
    add_library(production_core SHARED production_core.cpp)
    target_compile_features(production_core PRIVATE cxx_std_17)
    target_compile_options(production_core PRIVATE -Wall -Wextra -Wformat=2 -Werror)

    # Phase 2 has no JNI/UI entry points. Check its public headers on every APK
    # build, so errors in otherwise unused code can no longer remain hidden.
    add_library(phase2_compile_check OBJECT tests/phase2_header_compile.cpp)
    target_compile_features(phase2_compile_check PRIVATE cxx_std_17)
    target_compile_options(phase2_compile_check PRIVATE -Wall -Wextra -Wformat=2 -Werror -pedantic-errors)
    add_dependencies(production_core phase2_compile_check)

    option(NATIVESIM_BUILD_PHASE2_TESTS "Build native POSIX regression executables" OFF)
    if(NATIVESIM_BUILD_PHASE2_TESTS)
        foreach(component security file session)
            add_executable(phase2_${component}_tests tests/phase2_${component}_test.cpp)
            target_include_directories(phase2_${component}_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
            target_compile_features(phase2_${component}_tests PRIVATE cxx_std_17)
            target_compile_options(phase2_${component}_tests PRIVATE -Wall -Wextra -Wformat=2 -Werror -pedantic-errors)
        endforeach()
        target_link_options(phase2_file_tests PRIVATE -Wl,--wrap=read -Wl,--wrap=write -Wl,--wrap=readdir -Wl,--wrap=fstatat)
    endif()
else()
    enable_testing()
    find_package(Threads REQUIRED)
    add_executable(native_diagnostic_tests tests/native_diagnostic_test.cpp)
    target_include_directories(native_diagnostic_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_features(native_diagnostic_tests PRIVATE cxx_std_17)
    target_link_libraries(native_diagnostic_tests PRIVATE Threads::Threads)
    if(MSVC)
        target_compile_options(native_diagnostic_tests PRIVATE /W4 /WX)
    else()
        target_compile_options(native_diagnostic_tests PRIVATE -Wall -Wextra -Werror)
    endif()
    add_test(NAME native_diagnostic_tests COMMAND native_diagnostic_tests)

    add_executable(phase2_security_tests tests/phase2_security_test.cpp)
    target_include_directories(phase2_security_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_features(phase2_security_tests PRIVATE cxx_std_17)
    target_link_libraries(phase2_security_tests PRIVATE Threads::Threads)
    if(MSVC)
        target_compile_options(phase2_security_tests PRIVATE /W4 /WX)
    else()
        target_compile_options(phase2_security_tests PRIVATE -Wall -Wextra -Wformat=2 -Werror -pedantic-errors)
    endif()
    add_test(NAME phase2_security_tests COMMAND phase2_security_tests)
endif()
